class spi_agent_top extends uvm_env;
	`uvm_component_utils(spi_agent_top)

	spi_agent agnth[];
	env_config m_cfg;

	function new(string name ="spi_agent_top", uvm_component parent);
		super.new(name,parent);
	endfunction
	
	function void build_phase(uvm_phase phase);
		if(!uvm_config_db #(env_config)::get(this,"","env_config",m_cfg))
			`uvm_fatal(get_type_name(),"Error in getting configuration");

		if(m_cfg.has_spiagent) begin
			agnth = new[m_cfg.no_of_agents];
			foreach(agnth[i]) begin
				uvm_config_db #(spi_config)::set(this,$sformatf("agnth[%0d]*",i),"spi_config",m_cfg.spi_h[i]);
				agnth[i]=spi_agent::type_id::create($sformatf("agnth[%0d]",i),this);
			end
		end
		super.build_phase(phase);
	endfunction
endclass
