class virtual_sequence_base extends uvm_sequence #(uvm_sequence_item);

	`uvm_object_utils(virtual_sequence_base)

	env_config m_cfg;

	apb_sequencer apb_seqrh[];
	spi_sequencer spi_seqrh[];

	virtual_sequencer vsqrh;

//	rand uvm_reg_data_t data;
	uvm_status_e status;

	function new(string name="virtual_sequence_base");
		super.new(name);

	endfunction


	task body();

		if(!uvm_config_db #(env_config)::get(null,get_full_name(),"env_config",m_cfg))
			`uvm_fatal(get_type_name(),"Error in getting configuration")


		apb_seqrh = new[m_cfg.no_of_agents];
		spi_seqrh = new[m_cfg.no_of_agents];

		assert($cast(vsqrh,m_sequencer))
		else
			`uvm_error(get_type_name(),"Error in casting")


		foreach(apb_seqrh[i]) begin
			apb_seqrh[i] = vsqrh.apb_seqrh[i];
		end

		foreach(spi_seqrh[i]) begin
			spi_seqrh[i] = vsqrh.spi_seqrh[i];
		end
	endtask
endclass


class reset_virtual_sequence extends virtual_sequence_base;

	`uvm_object_utils(reset_virtual_sequence)

	apb_rst_seqs apb_seqs;
	function new(string name="reset_virtual_sequence");
		super.new(name);
	endfunction


	task body();
		super.body();
		apb_seqs = apb_rst_seqs::type_id::create("apb_seqs");
		apb_seqs.start(apb_seqrh[0]);
	endtask
endclass
