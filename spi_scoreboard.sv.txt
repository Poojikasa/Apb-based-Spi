class scoreboard extends uvm_scoreboard;

	`uvm_component_utils(scoreboard)
	env_config m_cfg;
	uvm_tlm_analysis_fifo #(apb_xtn) fifo_apb[];
	uvm_tlm_analysis_fifo #(spi_xtn) fifo_spi[];
	bit [7:0] data;
	uvm_status_e status;
	spi_reg_block spi_rg_blk;
	apb_xtn apb_xtnh;
	spi_xtn spi_xtnh;
	static int data_verified_cnt = 0;
	bit reset_case;
	bit [1:0]low_pwr_case;
	bit [7:0] cr1,cr2,br,sr,dr; 
	function new(string name="scoreboard", uvm_component parent);
		super.new(name,parent);
	endfunction

	function void build_phase(uvm_phase phase);
		if(!uvm_config_db #(env_config)::get(this,"","env_config",m_cfg))
			`uvm_fatal(get_type_name(),"Error in getting configuration")

		fifo_apb = new[m_cfg.no_of_agents];
		fifo_spi = new[m_cfg.no_of_agents];
		
		foreach(fifo_apb[i]) begin
			fifo_apb[i] = new($sformatf("fifo_apb[%0d]",i),this);
		end
	
		foreach(fifo_spi[i]) begin
			fifo_spi[i] = new($sformatf("fifo_spi[%0d]",i),this);
		end
	endfunction

	function void connect_phase(uvm_phase phase);
		spi_rg_blk = m_cfg.spi_rg_blk;
	endfunction


	task run_phase(uvm_phase phase);
		fork
		  begin 
		   forever
		   	fifo_apb[0].get(apb_xtnh);
			compare_data1(apb_xtnh);
		   end

		   begin
		    forever
			fifo_spi[0].get(spi_xtnh);
			compare_data(apb_xtnh);
		    end
		join
	endtask


	task compare_data(apb_xtn xtnh);
		wait(spi_xtnh != null);
		wait(apb_xtnh != null);
		if(apb_xtnh.PWRITE && apb_xtnh.PADDR == 3'b101) begin
			if(apb_xtnh.PWDATA == spi_xtnh.mosi)
				`uvm_info(get_type_name(),"succesfull verification", UVM_LOW)
			else
				`uvm_info(get_type_name(),"Unsuccessfull verification", UVM_LOW)
		end
	endtask

	        task compare_data1(apb_xtn a_xtn);
        /*comparision logic to verify
                1.miso data and PRDTA
                2.register in reset condition
                3.low power mode*/
              if(!uvm_config_db#(bit)::get(this,"","bit",reset_case))
                        `uvm_fatal(get_type_name(),"get the fatal")
                if(!uvm_config_db#(bit[1:0])::get(this,"","bit[1:0]",low_pwr_case))
                        `uvm_fatal(get_type_name(),"get the fatal")
                if(reset_case)
                        begin
                                this.spi_rg_blk.cr1.read(status,cr1,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                this.spi_rg_blk.cr2.read(status,cr2,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                this.spi_rg_blk.br.read(status,br,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                this.spi_rg_blk.sr.read(status,sr,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                this.spi_rg_blk.dr.read(status,dr,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                if((cr1==8'b0000_0100) && (cr2==8'b0000_0000) && (br==8'b0000_0000) && (sr==8'b0010_0000) && (dr==8'b0000_0000))
                                        `uvm_info(get_type_name(),"reset comparision is successfull",UVM_LOW)
                                else
                                        `uvm_error(get_type_name(),"reset comparision is failed")
                                `uvm_info(get_type_name(),$sformatf("the reset values of register are -- cr1=%0b,cr2=%0b,br=%0b,sr=%0b,dr=%0b",cr1,cr2,br,sr,dr),UVM_LOW)
                                data_verified_cnt++;
                        end
                if(low_pwr_case==2'b01)
                        begin
                                if((!a_xtn.PWRITE) && (a_xtn.PADDR==3'b101))
                                        begin
                                                this.spi_rg_blk.dr.read(status,dr,.path(UVM_BACKDOOR),.map(spi_rg_blk.spi_reg_map));
                                                $display("################################## scoreboard report #####################");
                                                if(a_xtn.PRDATA == dr)
                                                        `uvm_info(get_type_name(),"low_pwr_case  comparision is successfull ",UVM_LOW )
                                                else
                                                        `uvm_error(get_type_name(),"low_pwr_case  comparision is failed")
                                                $display("##########################################################################");
                                        end
                        end
                else
                        begin
                                wait(spi_xtnh!=null);
                                if((!a_xtn.PWRITE) && (a_xtn.PADDR==3'b101)) begin
					if(a_xtn.PRDATA == spi_xtnh.miso)
						`uvm_info(get_type_name(),"Comparison is succesfull",UVM_LOW)
					else
						`uvm_info(get_type_name(),"Unsuccesfull comparison",UVM_LOW)
				end
                        end
        endtask

		
			

endclass
