class spi_monitor extends uvm_monitor;

	`uvm_component_utils(spi_monitor)
	uvm_analysis_port #(spi_xtn) mon_port;
	spi_config sp_cfg;
	virtual spi_if.SPI_MON vif;

	bit [7:0]ctrl;
	bit cphase;
	bit cpol;
	bit lsbfe;
	function new(string name="spi_monitor", uvm_component parent);
		super.new(name,parent);
		mon_port = new("mon_port",this);
	endfunction

	 function void build_phase(uvm_phase phase);
                if(!uvm_config_db #(spi_config)::get(this,"","spi_config",sp_cfg))
                        `uvm_fatal(get_type_name(),"Error in getting configuration")

        endfunction


        function void connect_phase(uvm_phase phase);
                vif = sp_cfg.vif;
        endfunction


	task run_phase(uvm_phase phase);
		forever
			collect_data();
	endtask


	task collect_data();
		spi_xtn xtnh;
		xtnh = spi_xtn::type_id::create("xtnh");
		if(!uvm_config_db #(bit[7:0])::get(this,"","bit[7:0]",ctrl)) begin
                        `uvm_fatal(get_type_name(),"get the fatal")
		end
                          `uvm_info(get_type_name(),$sformatf("the drive data is %0b",ctrl),UVM_LOW)
                         cphase=ctrl[2];
                         cpol=ctrl[3];
                         lsbfe=ctrl[0];

		@(vif.spi_mon_cb);
		begin
			if(lsbfe) begin
				
				for(int i=0; i<=7; i++) begin
					if( ((!cpol) && (!cphase)) || ((cpol) && (cphase)))begin
						@(posedge vif.spi_mon_cb.SCLK)
						begin
							xtnh.miso[i] = vif.spi_mon_cb.MISO;
							xtnh.mosi[i] = vif.spi_mon_cb.MOSI;
							xtnh.ss = vif.spi_mon_cb.SS_n;
							xtnh.spi_interrupt_request = vif.spi_mon_cb.spi_int_req;
						end
					end

					else begin
						
						@(negedge vif.spi_mon_cb.SCLK)
						begin
							xtnh.miso[i] = vif.spi_mon_cb.MISO;
							xtnh.mosi[i] = vif.spi_mon_cb.MOSI;
							xtnh.ss = vif.spi_mon_cb.SS_n;
							xtnh.spi_interrupt_request = vif.spi_mon_cb.spi_int_req;
						end
					end
				end
			end

			else begin
				
				for(int i=7; i>=0; i--) begin
					if( ((!cpol) && (!cphase)) || ((cpol) && (cphase)))begin
						@(posedge vif.spi_mon_cb.SCLK)
						begin
							xtnh.miso[i] = vif.spi_mon_cb.MISO;
							xtnh.mosi[i] = vif.spi_mon_cb.MOSI;
							xtnh.ss = vif.spi_mon_cb.SS_n;
							xtnh.spi_interrupt_request = vif.spi_mon_cb.spi_int_req;
						end
					end

					else begin
						
						@(negedge vif.spi_mon_cb.SCLK)
						begin
							xtnh.miso[i] = vif.spi_mon_cb.MISO;
							xtnh.mosi[i] = vif.spi_mon_cb.MOSI;
							xtnh.ss = vif.spi_mon_cb.SS_n;
							xtnh.spi_interrupt_request = vif.spi_mon_cb.spi_int_req;
						end
					end
				end
			end
		end
		`uvm_info(get_type_name(),$sformatf("Printing from spi monitor %s",xtnh.sprint()),UVM_LOW)
	endtask
endclass
